#Function to match two sets of water level peak data by time, with a 4 hour buffer. Inputs two data frames,
#returns a single data frame. 
match.dates = function(site_peaks, NOAA_peaks){
  out = data.frame(matrix(nrow=nrow(site_peaks),ncol=5))
  site_peaks = as.data.frame(site_peaks)
  site_peaks[,datetime] = as.POSIXct(strptime(site_peaks[,datetime], "%Y-%m-%d %H:%M:%S")) #reads in Date Time
  NOAA_peaks[,1] = as.POSIXct(strptime(NOAA_peaks[,1], "%Y-%m-%d %H:%M")) #reads in Date Time
  NOAA_peaks = NOAA_peaks[complete.cases(NOAA_peaks),]  #removes missing data from NOAA table
  count =1  # 3926
 # STOP=F
  for(i in 1:nrow(site_peaks)){
    found = F    
    while(count < nrow(NOAA_peaks) & found==F ){
     if(abs(as.numeric(NOAA_peaks[count,1]) - as.numeric(site_peaks[[i,datetime]])) < 10800){   #10800 seconds is 4 hours
     #  STOP=T
       out[i,] = c(strftime(NOAA_peaks[count,1]), NOAA_peaks[count,2:3], strftime(site_peaks[i, datetime]),as.matrix(site_peaks[[i,baro_wl_level]]) )
       found=T  
     } else {
       count = count + 1
     }            
    }   
  }
  colnames(out) = c("NOAA_DT","NOAA_Level_NAVD_M","Tide","Site_DT","Site_Level_NAVD_M")
  out2 = out[complete.cases(out),]
  return(out2)
}
